CREATE TABLE JOBURI (
cod_job NUMBER(3),
nume_job VARCHAR2(25) NOT NULL,
salariu_minim NUMBER(8, 2),
salariu_maxim NUMBER(8, 2),
CONSTRAINT pk_joburi PRIMARY KEY (cod_job)
);

CREATE TABLE ANGAJATI (
cod_angajat NUMBER(5),
nume VARCHAR2(25) NOT NULL,
prenume VARCHAR2(25) NOT NULL,
salariu NUMBER(8, 2),
data_angajare DATE DEFAULT SYSDATE,
cnp CHAR(13),
cod_job NUMBER(3),
CONSTRAINT pk_angajati PRIMARY KEY (cod_angajat),
CONSTRAINT fk_angajati_joburi FOREIGN KEY (cod_job) REFERENCES JOBURI(cod_job),
CONSTRAINT uq_angajati_cnp UNIQUE (cnp)
);

CREATE TABLE SOFERI (
cod_angajat NUMBER(5),
expirare_permis DATE NOT NULL,
CONSTRAINT pk_soferi PRIMARY KEY (cod_angajat),
CONSTRAINT fk_soferi_angajati FOREIGN KEY (cod_angajat) REFERENCES ANGAJATI(cod_angajat)
);

CREATE TABLE TIPURI_VEHICUL (
cod_tip_vehicul NUMBER(3),
tip_vehicul VARCHAR2(25) NOT NULL,
categorie_permis VARCHAR2(2),
infrastructura VARCHAR2(25),
CONSTRAINT pk_tipuri_vehicul PRIMARY KEY (cod_tip_vehicul)
);

CREATE TABLE CATEGORII_PERMISE (
cod_angajat NUMBER(5),
cod_tip_vehicul NUMBER(3),
data_obtinere DATE,
CONSTRAINT pk_categorii_permise PRIMARY KEY (cod_angajat, cod_tip_vehicul),
CONSTRAINT fk_cat_soferi FOREIGN KEY (cod_angajat) REFERENCES SOFERI(cod_angajat),
CONSTRAINT fk_cat_tipuri FOREIGN KEY (cod_tip_vehicul) REFERENCES TIPURI_VEHICUL(cod_tip_vehicul)
);

CREATE TABLE VEHICULE (
cod_vehicul NUMBER(5),
model VARCHAR2(50),
an_fabricare NUMBER(4),
nr_inmatriculare VARCHAR2(10),
cod_tip_vehicul NUMBER(3),
CONSTRAINT pk_vehicule PRIMARY KEY (cod_vehicul),
CONSTRAINT uq_nr_inmatriculare UNIQUE (nr_inmatriculare),
CONSTRAINT fk_tipuri_vehicul FOREIGN KEY (cod_tip_vehicul) REFERENCES TIPURI_VEHICUL(cod_tip_vehicul)
);

CREATE TABLE TIP_LINIE (
cod_tip_linie NUMBER(2),
tip_linie VARCHAR2(25) NOT NULL,
CONSTRAINT pk_tip_linie PRIMARY KEY (cod_tip_linie)
);

CREATE TABLE LINII (
cod_linie NUMBER(4),
numar_linie VARCHAR2(10) NOT NULL,
km NUMBER(5, 2),
cod_tip_linie NUMBER(3),
CONSTRAINT pk_linii PRIMARY KEY (cod_linie),
CONSTRAINT fk_linii_tip FOREIGN KEY (cod_tip_linie) REFERENCES TIP_LINIE(cod_tip_linie)
);

CREATE TABLE STATII (
cod_statie NUMBER(5),
nume_statie VARCHAR2(100) NOT NULL,
adresa VARCHAR2(100),
punct_vanzare CHAR(2),
CONSTRAINT pk_statii PRIMARY KEY (cod_statie),
CONSTRAINT ck_punct_vanzare CHECK (punct_vanzare IN ('DA', 'NU'))

);

CREATE TABLE TRASEU (
cod_linie NUMBER(4),
cod_statie NUMBER(5),
ordine NUMBER(3),
CONSTRAINT pk_traseu PRIMARY KEY (cod_linie, cod_statie),
CONSTRAINT fk_traseu_linii FOREIGN KEY (cod_linie) REFERENCES LINII(cod_linie),
CONSTRAINT fk_traseu_statii FOREIGN KEY (cod_statie) REFERENCES STATII(cod_statie)
);

CREATE TABLE TURE (
cod_linie NUMBER(4),
cod_angajat NUMBER(5),
cod_vehicul NUMBER(5),
tura NUMBER(1),
data_tura DATE,
CONSTRAINT pk_ture PRIMARY KEY (cod_linie, cod_angajat, cod_vehicul, data_tura),
CONSTRAINT fk_ture_linii FOREIGN KEY (cod_linie) REFERENCES LINII(cod_linie),
CONSTRAINT fk_ture_soferi FOREIGN KEY (cod_angajat) REFERENCES SOFERI(cod_angajat),
CONSTRAINT fk_ture_vehicule FOREIGN KEY (cod_vehicul) REFERENCES VEHICULE(cod_vehicul)
);

CREATE TABLE CLIENTI (
cod_client NUMBER(8),
nume VARCHAR2(25),
prenume VARCHAR2(25),
cnp CHAR(13),
CONSTRAINT pk_clienti PRIMARY KEY (cod_client),
CONSTRAINT uq_clienti_cnp UNIQUE (cnp)
);

CREATE TABLE NIVEL_ACCES (
cod_abonament NUMBER(8),
cod_tip_linie NUMBER(3),
CONSTRAINT pk_nivel_acces PRIMARY KEY (cod_abonament, cod_tip_linie),
CONSTRAINT fk_acces_ab FOREIGN KEY (cod_abonament) REFERENCES ABONAMENTE(cod_abonament),
CONSTRAINT fk_acces_tip_linie FOREIGN KEY (cod_tip_linie) REFERENCES TIP_LINIE(cod_tip_linie)
);

CREATE TABLE ABONAMENTE (
cod_abonament NUMBER(8),
cod_client NUMBER(8),
data_activare DATE DEFAULT SYSDATE NOT NULL,
data_expirare DATE NOT NULL,
CONSTRAINT pk_abonamente PRIMARY KEY (cod_abonament),
CONSTRAINT fk_ab_clienti FOREIGN KEY (cod_client) REFERENCES CLIENTI(cod_client),
CONSTRAINT ck_perioada_valida CHECK (data_expirare > data_activare)
);
